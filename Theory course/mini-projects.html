Student Management System — Student Model

import java.io.Serializable; public class Student implements Serializable { private static final long serialVersionUID = 1L; private int id; private String name; private int age; private double marks; public Student(int id, String name, int age, double marks) { this.id = id; this.name = name; this.age = age; this.marks = marks; } public int getId() { return id; } public String getName() { return name; } public int getAge() { return age; } public double getMarks() { return marks; } public void setMarks(double m) { this.marks = m; } public String getGrade() { if (marks >= 90) return "A"; else if (marks >= 75) return "B"; else if (marks >= 60) return "C"; else if (marks >= 50) return "D"; else return "F"; } @Override public String toString() { return String.format("%-5d %-15s %-5d %-8.1f %s", id, name, age, marks, getGrade()); } }

StudentManager — Logic and File Storage

import java.io.*; import java.util.*; public class StudentManager { private ArrayList students = new ArrayList<>(); private static final String FILE = "students.ser"; private int nextId = 1; public StudentManager() { load(); } public void add(String name, int age, double marks) { students.add(new Student(nextId++, name, age, marks)); save(); System.out.println("Student added."); } public void viewAll() { if (students.isEmpty()) { System.out.println("No students found."); return; } System.out.printf("%-5s %-15s %-5s %-8s %s%n", "ID", "Name", "Age", "Marks", "Grade"); System.out.println("-".repeat(42)); students.forEach(System.out::println); } public void search(String name) { students.stream() .filter(s -> s.getName().equalsIgnoreCase(name)) .forEach(System.out::println); } public void updateMarks(int id, double marks) { students.stream() .filter(s -> s.getId() == id) .findFirst() .ifPresentOrElse(s -> { s.setMarks(marks); save(); System.out.println("Marks updated."); }, () -> System.out.println("Student not found.")); } public void delete(int id) { boolean removed = students.removeIf(s -> s.getId() == id); if (removed) { save(); System.out.println("Deleted."); } else System.out.println("Student not found."); } public void topStudents() { System.out.println("=== Grade A Students ==="); students.stream() .filter(s -> s.getGrade().equals("A")) .sorted((a, b) -> Double.compare(b.getMarks(), a.getMarks())) .forEach(System.out::println); } private void save() { try (ObjectOutputStream oos = new ObjectOutputStream( new FileOutputStream(FILE))) { oos.writeObject(students); oos.writeInt(nextId); } catch (IOException e) { e.printStackTrace(); } } @SuppressWarnings("unchecked") private void load() { File f = new File(FILE); if (!f.exists()) return; try (ObjectInputStream ois = new ObjectInputStream( new FileInputStream(FILE))) { students = (ArrayList) ois.readObject(); nextId = ois.readInt(); } catch (IOException | ClassNotFoundException e) { e.printStackTrace(); } } }

Student System — Main Menu

import java.util.Scanner; public class Main { public static void main(String[] args) { StudentManager manager = new StudentManager(); Scanner sc = new Scanner(System.in); while (true) { System.out.println("\n=== Student Management System ==="); System.out.println("1. Add Student"); System.out.println("2. View All"); System.out.println("3. Search by Name"); System.out.println("4. Update Marks"); System.out.println("5. Delete Student"); System.out.println("6. Top Students"); System.out.println("7. Bank Management System — Model

import java.io.Serializable; import java.util.ArrayList; public class BankAccount implements Serializable { private static final long serialVersionUID = 1L; private int accountNo; private String owner; private double balance; private ArrayList history = new ArrayList<>(); public BankAccount(int accountNo, String owner, double deposit) { this.accountNo = accountNo; this.owner = owner; this.balance = deposit; log("Account opened. Balance: " + deposit); } public int getAccountNo() { return accountNo; } public String getOwner() { return owner; } public double getBalance() { return balance; } public void deposit(double amount) { if (amount <= 0) { System.out.println("Invalid amount."); return; } balance += amount; log("Deposited: " + amount + " | Balance: " + balance); System.out.printf("Deposited %.2f | New Balance: %.2f%n", amount, balance); } public boolean withdraw(double amount) { if (amount <= 0) { System.out.println("Invalid amount."); return false; } if (balance < amount) { System.out.println("Insufficient funds."); return false; } balance -= amount; log("Withdrew: " + amount + " | Balance: " + balance); System.out.printf("Withdrew %.2f | New Balance: %.2f%n", amount, balance); return true; } public void printHistory() { System.out.println("=== History: " + owner + " ==="); history.forEach(h -> System.out.println(" " + h)); } private void log(String entry) { history.add(entry); } @Override public String toString() { return String.format("Acc#%-6d %-15s Balance: $%.2f", accountNo, owner, balance); } }

Bank Class

import java.io.*; import java.util.*; public class Bank { private HashMap accounts = new HashMap<>(); private static final String FILE = "bank.ser"; private int nextNo = 1001; public Bank() { load(); } public void createAccount(String owner, double deposit) { BankAccount acc = new BankAccount(nextNo++, owner, deposit); accounts.put(acc.getAccountNo(), acc); save(); System.out.println("Account created: #" + acc.getAccountNo()); } public BankAccount get(int no) { return accounts.getOrDefault(no, null); } public void transfer(int from, int to, double amount) { BankAccount a = get(from), b = get(to); if (a == null || b == null) { System.out.println("Account not found."); return; } if (a.withdraw(amount)) { b.deposit(amount); save(); } } public void viewAll() { if (accounts.isEmpty()) { System.out.println("No accounts."); return; } accounts.values().forEach(System.out::println); } private void save() { try (ObjectOutputStream oos = new ObjectOutputStream( new FileOutputStream(FILE))) { oos.writeObject(accounts); oos.writeInt(nextNo); } catch (IOException e) { e.printStackTrace(); } } @SuppressWarnings("unchecked") private void load() { File f = new File(FILE); if (!f.exists()) return; try (ObjectInputStream ois = new ObjectInputStream( new FileInputStream(FILE))) { accounts = (HashMap) ois.readObject(); nextNo = ois.readInt(); } catch (IOException | ClassNotFoundException e) { e.printStackTrace(); } } }

Inventory System — Product Model

public class Product { private int id; private String name; private double price; private int qty; public Product(int id, String name, double price, int qty) { this.id = id; this.name = name; this.price = price; this.qty = qty; } public int getId() { return id; } public String getName() { return name; } public int getQty() { return qty; } public void setQty(int q) { this.qty = q; } public String toCSV() { return id + "," + name + "," + price + "," + qty; } public static Product fromCSV(String line) { String[] p = line.split(","); return new Product(Integer.parseInt(p[0]), p[1], Double.parseDouble(p[2]), Integer.parseInt(p[3])); } @Override public String toString() { return String.format("%-5d %-15s $%-8.2f Qty: %d", id, name, price, qty); } }

Inventory — CSV Save/Load

import java.io.*; import java.util.*; public class Inventory { private ArrayList products = new ArrayList<>(); private static final String FILE = "inventory.csv"; private int nextId = 1; public Inventory() { load(); } public void add(String name, double price, int qty) { products.add(new Product(nextId++, name, price, qty)); save(); System.out.println("Added: " + name); } public void viewAll() { if (products.isEmpty()) { System.out.println("No products."); return; } products.forEach(System.out::println); } public void sell(int id, int qty) { products.stream().filter(p -> p.getId() == id).findFirst() .ifPresentOrElse(p -> { if (p.getQty() < qty) System.out.println("Not enough stock."); else { p.setQty(p.getQty() - qty); save(); System.out.println("Sold. Remaining: " + p.getQty()); } }, () -> System.out.println("Product not found.")); } public void lowStock(int limit) { products.stream().filter(p -> p.getQty() < limit) .forEach(p -> System.out.println("LOW: " + p)); } private void save() { try (BufferedWriter bw = new BufferedWriter(new FileWriter(FILE))) { bw.write(nextId + "\n"); for (Product p : products) bw.write(p.toCSV() + "\n"); } catch (IOException e) { e.printStackTrace(); } } private void load() { File f = new File(FILE); if (!f.exists()) return; try (BufferedReader br = new BufferedReader(new FileReader(FILE))) { nextId = Integer.parseInt(br.readLine().trim()); String line; while ((line = br.readLine()) != null) if (!line.trim().isEmpty()) products.add(Product.fromCSV(line)); } catch (IOException e) { e.printStackTrace(); } } }
